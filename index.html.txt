        function renderDateStrip(currentDateObj) {
            const dateStrip = document.getElementById('m-date-strip');
            dateStrip.innerHTML = '';

            const currentYear = currentDateObj.getFullYear();
            const currentMonth = currentDateObj.getMonth();
            const daysInMonth = new Date(currentYear, currentMonth + 1, 0).getDate();

            for (let i = 1; i <= daysInMonth; i++) {
                const loopDate = new Date(currentYear, currentMonth, i);
                const dayAbbr = dayNames[loopDate.getDay()].substring(0, 3);
                const yyyy = currentYear;
                const mm = String(currentMonth + 1).padStart(2, '0');
                const dd = String(i).padStart(2, '0');
                const dateVal = `${yyyy}-${mm}-${dd}`;

                const isToday = (i === currentDateObj.getDate());
                const isSelected = (dateVal === selectedDateStr);

                let bgClass = "bg-slate-100 text-slate-700 hover:bg-slate-200 border-slate-200";
                if (isSelected) {
                    bgClass = "bg-sky-600 text-white font-extrabold border-sky-600 shadow";
                } else if (isToday) {
                    bgClass = "bg-amber-100 text-amber-900 border-amber-300 font-extrabold";
                }

                dateStrip.innerHTML += `
                    <button onclick="selectDatePill('${dateVal}')" class="flex-shrink-0 px-3 py-1.5 rounded-2xl border text-center transition flex flex-col items-center justify-center min-w-[50px] ${bgClass}">
                        <span class="text-[9px] uppercase font-bold tracking-wider">${dayAbbr}</span>
                        <span class="text-xs font-black">${i}</span>

        // Signature Canvas Logic
        let canvas, ctx, isDrawing = false;

        function initSignaturePad() {
            canvas = document.getElementById('signature-pad');
            if (!canvas) return;
            ctx = canvas.getContext('2d');
            ctx.strokeStyle = "#0f172a";
            ctx.lineWidth = 2;

            function getPos(e) {
                const rect = canvas.getBoundingClientRect();
                return {
                    x: (e.clientX || e.touches[0].clientX) - rect.left,
                    y: (e.clientY || e.touches[0].clientY) - rect.top
                };
            }

            canvas.addEventListener('mousedown', (e) => { isDrawing = true; const pos = getPos(e); ctx.beginPath(); ctx.moveTo(pos.x, pos.y); });
            canvas.addEventListener('mousemove', (e) => { if (!isDrawing) return; const pos = getPos(e); ctx.lineTo(pos.x, pos.y); ctx.stroke(); });
            canvas.addEventListener('mouseup', () => isDrawing = false);

            canvas.addEventListener('touchstart', (e) => { isDrawing = true; const pos = getPos(e); ctx.beginPath(); ctx.moveTo(pos.x, pos.y); });
            canvas.addEventListener('touchmove', (e) => { if (!isDrawing) return; const pos = getPos(e); ctx.lineTo(pos.x, pos.y); ctx.stroke(); e.preventDefault(); });
            canvas.addEventListener('touchend', () => isDrawing = false);
        }

        function clearSignature() {
            if (ctx && canvas) {
                ctx.clearRect(0, 0, canvas.width, canvas.height);
            }
        }

        function handleFormSubmit(e) {
            e.preventDefault();

            const pdfInput = document.getElementById('f-pdf');
            let pdfName = 'Surat_Undangan.pdf';
            let pdfData = null;

            if (pdfInput.files && pdfInput.files[0]) {
                pdfName = pdfInput.files[0].name;
                const reader = new FileReader();
                reader.onload = function(evt) {
                    pdfData = evt.target.result;
                    saveNewAgenda(pdfName, pdfData);
                };
                reader.readAsDataURL(pdfInput.files[0]);
            } else {
                saveNewAgenda(pdfName, pdfData);
            }
        }

        function sendToGoogleBackend(data) {
            if (!APPS_SCRIPT_WEB_APP_URL || APPS_SCRIPT_WEB_APP_URL.includes('YOUR_SCRIPT_ID_HERE')) return;

            fetch(APPS_SCRIPT_WEB_APP_URL, {
                method: 'POST',
                mode: 'no-cors',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ ...data, role: currentRole })
            }).then(() => console.log('Data tersimpan ke Google Sheets & Calendar'))
              .catch(err => console.error('Error Google Sync:', err));
        }

        function loadAgendasFromGoogleSheets() {
            if (!APPS_SCRIPT_WEB_APP_URL || APPS_SCRIPT_WEB_APP_URL.includes('YOUR_SCRIPT_ID_HERE')) return;

            fetch(APPS_SCRIPT_WEB_APP_URL)
                .then(res => res.json())
                .then(resData => {
                    if (resData.result === 'success' && resData.data && resData.data.length > 0) {
                        agendas = resData.data;
                        renderAgendas();
                    }
                })
                .catch(err => console.log('Google Sheets fetch bypass (demo mode)'));
        }
